// ── BlurGuard FP/FN Assessment — SW snippet ───────────────────────────────────
// Paste into SERVICE WORKER devtools console BEFORE clicking "Classify all".
// After cards settle, call fpTable() to print the assessment Markdown table.

(function() {
  if (self.__fpInstalled) { console.log('[fp] already installed'); return; }
  self.__fpInstalled = true;

  // ── Image corpus (mirrors test-fp.html IMAGES array) ──────────────────────
  // Update this if you change IDs or labels in the HTML file.
  const CORPUS = [
    { id: "safe-01", label: "Mountain landscape",              truth: "safe"     },
    { id: "safe-02", label: "Dog portrait (Yellow Lab)",       truth: "safe"     },
    { id: "safe-03", label: "Fruit & food display",            truth: "safe"     },
    { id: "safe-04", label: "Architecture (Colosseum)",        truth: "safe"     },
    { id: "safe-05", label: "Clothed male portrait",           truth: "safe"     },
    { id: "safe-06", label: "Forest path (nature)",            truth: "safe"     },
    { id: "safe-07", label: "Ocean at sunset",                 truth: "safe"     },
    { id: "safe-08", label: "City park (urban landscape)",     truth: "safe"     },
    { id: "safe-09", label: "Library / books",                 truth: "safe"     },
    { id: "safe-10", label: "Fashion (clothed runway)",        truth: "safe"     },
    { id: "edge-01", label: "Yoga (Ashtanga, tight sportswear)", truth: "edge"     },
    { id: "edge-02", label: "Botticelli Primavera (semi-clothed)", truth: "edge" },
    { id: "edge-03", label: "Botticelli Birth of Venus (nude)",    truth: "edge" },
    { id: "edge-04", label: "Michelangelo's David (sculpture)",    truth: "edge" },
    { id: "edge-05", label: "Mona Lisa (clothed, sanity check)",   truth: "edge" },
    { id: "nsfw-01", label: "[explicit 1]",                    truth: "explicit" },
    { id: "nsfw-02", label: "[explicit 2]",                    truth: "explicit" },
    { id: "nsfw-03", label: "[explicit 3]",                    truth: "explicit" },
    { id: "nsfw-04", label: "[borderline explicit]",           truth: "explicit" },
    { id: "nsfw-05", label: "[explicit 5]",                    truth: "explicit" },
  ];

  // id → corpus entry
  const byId = new Map(CORPUS.map(c => [c.id, c]));
  // id → captured verdict data
  const captured = new Map();

  // ── Intercept console.log ─────────────────────────────────────────────────
  // Log line format (from background.ts SW context — NOT offscreen doc context):
  //   [BlurGuard tfjs] id= verdict= block=
  //     scores={Porn:N,Hentai:N,Sexy:N,Neutral:N,Drawing:N}
  //     infMs=N queueMs=N decodeMs=N latencyMs=N
  //
  // NOTE: offscreen.ts logs as "[BlurGuard offscreen]" but those lines do NOT
  // contain verdict/scores and appear in the offscreen doc's OWN inspector, not
  // here. The SW console only receives "[BlurGuard tfjs]" lines from background.ts.

  const _orig = console.log.bind(console);
  console.log = (...args) => {
    _orig(...args);
    const s = args.join(' ');
    if (!s.includes('[BlurGuard tfjs]') || !s.includes('id=')) return;

    const id      = (s.match(/\bid=(\S+)/))?.[1];
    const verdict = (s.match(/verdict=(\w+)/))?.[1];
    const block   = s.includes('block=true');
    const porn    = parseFloat((s.match(/Porn:([\d.]+)/))?.[1] ?? '0');
    const hentai  = parseFloat((s.match(/Hentai:([\d.]+)/))?.[1] ?? '0');
    const sexy    = parseFloat((s.match(/Sexy:([\d.]+)/))?.[1] ?? '0');
    const neutral = parseFloat((s.match(/Neutral:([\d.]+)/))?.[1] ?? '0');
    const drawing = parseFloat((s.match(/Drawing:([\d.]+)/))?.[1] ?? '0');
    const infMs   = parseInt((s.match(/infMs=(\d+)/))?.[1] ?? '0', 10);

    if (id && byId.has(id)) {
      captured.set(id, { id, verdict, block, porn, hentai, sexy, neutral, drawing, infMs });
      _orig(`[fp] captured ${id} → verdict=${verdict} block=${block} Porn=${porn.toFixed(3)} Sexy=${sexy.toFixed(3)}`);
    }
  };

  // ── fpTable() — call after all images finish ──────────────────────────────
  self.fpTable = function() {
    const fmt = n => n.toFixed(3).replace(/^0/, '');  // .xxx
    const pct = n => (n * 100).toFixed(1) + '%';

    // ── Markdown table ────────────────────────────────────────────────────
    const rows = [];
    let fpCount = 0, fnCount = 0, tpCount = 0, tnCount = 0;

    rows.push('| # | ID | Label | Truth | Porn | Hentai | Sexy | Neutral | Drawing | Verdict | Block | Correct? | Error type |');
    rows.push('|---|----|----|----|----|----|----|----|----|----|----|----|----|');

    let n = 0;
    for (const c of CORPUS) {
      n++;
      const d = captured.get(c.id);
      if (!d) {
        const placeholder = c.id.startsWith('nsfw') ? 'no URL (placeholder)' : 'no data yet';
        rows.push(`| ${n} | ${c.id} | ${c.label} | ${c.truth} | — | — | — | — | — | — | — | — | ${placeholder} |`);
        continue;
      }

      // Correctness logic:
      //   safe  truth → correct if block=false
      //   edge  truth → no ground truth; note the verdict, no error count
      //   explicit truth → correct if block=true
      let correct = '—';
      let errorType = '';

      if (c.truth === 'safe') {
        if (!d.block) {
          correct = '✓';
          tnCount++;
        } else {
          correct = '✗';
          fpCount++;
          errorType = d.verdict === 'explicit'
            ? '🔴 FP (safe→explicit)'
            : '🟠 FP (safe→suggestive-blocked)';
        }
        // Non-blocked suggestive on a safe image: still worth flagging as a quality signal
        if (!d.block && d.verdict === 'suggestive') {
          errorType = '🟡 suggestive (not blocked)';
        }
      } else if (c.truth === 'explicit') {
        if (d.block) {
          correct = '✓';
          tpCount++;
        } else {
          correct = '✗';
          fnCount++;
          errorType = '🟣 FN (explicit→safe/not-blocked)';
        }
      } else {
        // edge — just report what the model said, no error classification
        correct = 'edge';
        errorType = d.block ? '⚠ blocked' : d.verdict !== 'safe' ? '⚠ flagged (not blocked)' : '';
      }

      rows.push(
        `| ${n} | ${c.id} | ${c.label} | ${c.truth} ` +
        `| ${fmt(d.porn)} | ${fmt(d.hentai)} | ${fmt(d.sexy)} ` +
        `| ${fmt(d.neutral)} | ${fmt(d.drawing)} ` +
        `| **${d.verdict}** | ${d.block ? '**YES**' : 'no'} ` +
        `| ${correct} | ${errorType} |`
      );
    }

    // ── Summary ───────────────────────────────────────────────────────────
    const n_safe     = CORPUS.filter(c => c.truth === 'safe').length;
    const n_explicit = CORPUS.filter(c => c.truth === 'explicit' && captured.has(c.id)).length;

    _orig('\n' + rows.join('\n'));
    _orig([
      '',
      '## FP/FN Summary',
      '',
      `- **False positives (safe→blocked):** ${fpCount} / ${n_safe}`,
      `- **False negatives (explicit→not-blocked):** ${fnCount} / ${n_explicit || '?'} (add explicit URLs to measure)`,
      `- **True negatives (safe correctly passed):** ${tnCount} / ${n_safe}`,
      `- **True positives (explicit correctly blocked):** ${tpCount} / ${n_explicit || '?'}`,
      '',
      '### Edge-case observations',
    ].join('\n'));

    // Print edge-case verdicts individually for manual annotation
    for (const c of CORPUS.filter(c => c.truth === 'edge')) {
      const d = captured.get(c.id);
      if (!d) { _orig(`  - ${c.id} (${c.label}): no data`); continue; }
      const dominant = ['Porn','Hentai','Sexy','Neutral','Drawing']
        .map(k => ({ k, v: d[k.toLowerCase()] }))
        .sort((a,b) => b.v - a.v)[0];
      _orig(`  - ${c.id} (${c.label}): dominant=${dominant.k} ${pct(dominant.v)}  verdict=${d.verdict}  block=${d.block}`);
    }

    _orig('\n[fp] Copy the table above → paste into your assessment doc / 2.4 tuning notes.');
  };

  _orig('[fp] installed. Classify images, then call fpTable().');
})();
