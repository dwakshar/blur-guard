// ── BlurGuard bench collector ─────────────────────────────────────────────────
// Paste into the SERVICE WORKER devtools console (chrome://extensions → BlurGuard → Service worker)
// BEFORE loading/reloading the test page.
// After all images settle, call bench() to print median / p95 / worst-case.

(function() {
  if (window.__bgBenchInstalled) { console.log('[bench] already installed'); return; }
  window.__bgBenchInstalled = true;

  const _inf = [], _queue = [], _decode = [], _lat = [];
  const _orig = console.log.bind(console);

  console.log = (...args) => {
    _orig(...args);
    const s = args.join(' ');
    // Matches: [BlurGuard offscreen] id=… verdict=… block=… scores={…} infMs=N queueMs=N decodeMs=N latencyMs=N
    const m = s.match(/infMs=(\d+)\s+queueMs=(\d+)\s+decodeMs=(\d+)\s+latencyMs=(\d+)/);
    if (m) {
      _inf.push(+m[1]);
      _queue.push(+m[2]);
      _decode.push(+m[3]);
      _lat.push(+m[4]);
    }
  };

  window.bench = function() {
    if (!_inf.length) { _orig('[bench] no data yet — run classification first'); return; }
    const sorted = arr => [...arr].sort((a, b) => a - b);
    const med  = arr => { const s = sorted(arr); return s[Math.floor(s.length / 2)]; };
    const p95  = arr => { const s = sorted(arr); return s[Math.floor(s.length * 0.95)] ?? s[s.length-1]; };
    const max  = arr => Math.max(...arr);
    const n    = _inf.length;
    _orig(
      `\n╔══ BlurGuard bench  n=${n} images ══════════════════════════════╗\n` +
      `║  inferenceMs (GPU work)    med=${med(_inf)}ms  p95=${p95(_inf)}ms  max=${max(_inf)}ms\n` +
      `║  queueWaitMs (serialise)   med=${med(_queue)}ms  p95=${p95(_queue)}ms  max=${max(_queue)}ms\n` +
      `║  decodeMs    (fetch+draw)  med=${med(_decode)}ms  p95=${p95(_decode)}ms  max=${max(_decode)}ms\n` +
      `║  latencyMs   (user-perc.)  med=${med(_lat)}ms  p95=${p95(_lat)}ms  max=${max(_lat)}ms\n` +
      `╠════════════════════════════════════════════════════════════════════╣\n` +
      `║  Throughput framing:  ~${med(_inf)}ms/image steady-state\n` +
      `║  Worst-case latency:  ${max(_lat)}ms  (last of ${n} queued)\n` +
      `║  Staircase check:     inferenceMs spread = ${max(_inf)-Math.min(..._inf)}ms ` +
              `(should be small; large → staircase not gone)\n` +
      `╚════════════════════════════════════════════════════════════════════╝`
    );
    _orig('[bench] Copy the block above → paste into README Performance section.');
  };

  _orig('[bench] installed. Run classification, then call bench().');
})();
